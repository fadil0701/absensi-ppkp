# Perintah untuk memperbaiki SatpelkesSeeder langsung di server
# Copy dan jalankan perintah-perintah berikut di server:

cd /var/www/html/absensi-ppkp

# Backup file terlebih dahulu
cp database/seeders/SatpelkesSeeder.php database/seeders/SatpelkesSeeder.php.backup

# Fix menggunakan sed (ganti koma dengan titik)
sed -i 's/-6,18191530/-6.18191530/g' database/seeders/SatpelkesSeeder.php
sed -i 's/106,82915070/106.82915070/g' database/seeders/SatpelkesSeeder.php
sed -i 's/-6,18195980/-6.18195980/g' database/seeders/SatpelkesSeeder.php
sed -i 's/106,82914410/106.82914410/g' database/seeders/SatpelkesSeeder.php

# Atau gunakan perintah berikut untuk fix semua sekaligus:
sed -i 's/\(latitude\|longitude\) => -\([0-9]\),\([0-9]*\)/\1 => -\2.\3/g' database/seeders/SatpelkesSeeder.php
sed -i 's/\(latitude\|longitude\) => \([0-9][0-9][0-9]\),\([0-9]*\)/\1 => \2.\3/g' database/seeders/SatpelkesSeeder.php

# Verifikasi perubahan
echo "Verifying changes..."
grep -n "latitude\|longitude" database/seeders/SatpelkesSeeder.php

# Setelah itu jalankan seeder
php artisan db:seed
